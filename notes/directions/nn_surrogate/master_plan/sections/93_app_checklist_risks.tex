\section{Checklist}
%-----------------------------------------------------------------------

\paragraph{Before any production run}
\begin{itemize}[leftmargin=2em]
  \ck Machine chosen for this member and recorded --- do not split a member
      across machines
  \ck (Port only) both pickups transferred, verified, second copies kept
  \ck (Port only) \code{input\_binaries/} and
      \code{input\_adj\_binaries/ones\_64b.bin} transferred
  \ck (Port only) Java runtime confirmed for Tapenade; \code{NERSC\_ACCOUNT}
      set; walltime set
  \ck Build completes; Tapenade adjoint builds
  \ck Forward build verified: 10-year run from rest reproduces the spin-up
      bit-identically
  \ck Adjoint build verified against reference run 28486 output to round-off
  \ck \textbf{Not} the gradient check --- it cannot pass at its current point
      (Appendix~\ref{app:grdchk}). Do not wait on it
  \ck Benchmark done; adjoint/forward ratio confirmed; \textbf{gradient-check
      saving measured}; figure reported
  \ck Node packing tested (port only)
  \ck Seven $\kv$ files generated, each $2{,}908{,}224$ bytes
  \ck Checkpoint at $\approx$2170 located;
      \code{nTimeSteps = 3162240 $-$ nIter0} set accordingly
  \ck \code{useGrdchk = .FALSE.} set for members
  \ck \code{pChkptFreq} decision made --- members only, leaving the spin-up's
      restarts alone
  \ck Forward diagnostic cadence decision made (Appendix~\ref{app:namelists})
  \ck Namelist variants placed in \code{input*/variants/kappa\_v\_ensemble/},
      not directly in \code{input*/} --- anything there is staged into every run
  \ck \code{nIter0} and the pickup symlink edited together in each submit script
\end{itemize}

\paragraph{After the pilot}
\begin{itemize}[leftmargin=2em]
  \ck Overturning drift at year 10 assessed
  \ck Convective activity diagnosed
  \ck Sensitivity patterns compared against the reference
  \ck Reviewed with collaborators before committing the remainder
\end{itemize}

\paragraph{Ongoing}
\begin{itemize}[leftmargin=2em]
  \ck \code{run\_timing.txt} retained for every run
  \ck Any per-member deviation in timestep or solver settings recorded
\end{itemize}
